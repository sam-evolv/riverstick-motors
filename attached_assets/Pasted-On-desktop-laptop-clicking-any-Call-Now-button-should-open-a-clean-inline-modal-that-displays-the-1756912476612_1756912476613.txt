Detect device type robustly: treat as mobile if navigator.userAgent contains Mobi or Android or if window.matchMedia("(pointer:coarse)") is true. Otherwise treat as desktop.

Visuals must match site theme: subtle overlay, rounded card, same font stack, no layout shifts, keyboard accessible, ESC to close, click-outside to close, focus trap within modal.

Idempotent: safe to run multiple times.

Do not change copy, layout, colors, or behavior anywhere else.

JavaScript (add to the main JS bundle or a new call-modal.js and include it):

(function(){
  const isMobile = () => /Mobi|Android/i.test(navigator.userAgent) || window.matchMedia('(pointer:coarse)').matches;
  const ensureModal = () => {
    if (document.getElementById('callModal')) return document.getElementById('callModal');
    const wrapper = document.createElement('div');
    wrapper.innerHTML = `
      <div id="callModal" aria-hidden="true" role="dialog" aria-modal="true">
        <div class="call-modal__backdrop" data-close></div>
        <div class="call-modal__card" role="document" tabindex="-1">
          <button class="call-modal__close" aria-label="Close" data-close>×</button>
          <h2 class="call-modal__title">Call Riverstick Motors</h2>
          <p class="call-modal__number" data-number="+353214771362">(021) 477 1362</p>
          <div class="call-modal__actions">
            <button class="call-modal__copy">Copy number</button>
          </div>
          <p class="call-modal__hint">Or type it into your phone now.</p>
        </div>
      </div>`;
    document.body.appendChild(wrapper.firstElementChild);
    return document.getElementById('callModal');
  };
  const openModal = () => {
    const m = ensureModal();
    m.setAttribute('aria-hidden','false');
    const card = m.querySelector('.call-modal__card');
    card.focus();
    document.body.style.overflow='hidden';
  };
  const closeModal = () => {
    const m = document.getElementById('callModal');
    if (!m) return;
    m.setAttribute('aria-hidden','true');
    document.body.style.overflow='';
  };
  const attachEvents = () => {
    const modal = ensureModal();
    modal.addEventListener('click', (e)=>{
      if (e.target.hasAttribute('data-close')) closeModal();
    });
    document.addEventListener('keydown', (e)=>{ if (e.key === 'Escape') closeModal(); });
    modal.querySelector('.call-modal__copy').addEventListener('click', async ()=>{
      const number = modal.querySelector('.call-modal__number').dataset.number || '+353214771362';
      try { await navigator.clipboard.writeText(number); } catch(e) { /* fallback noop */ }
    });
  };
  const interceptDesktopClicks = () => {
    document.querySelectorAll('a[href^="tel:"], .js-call-now').forEach(el=>{
      el.classList.add('js-call-now');
      el.addEventListener('click', (evt)=>{
        if (isMobile()) return; // allow default tel: on mobile
        evt.preventDefault();
        openModal();
      }, {passive:false});
    });
  };
  document.addEventListener('DOMContentLoaded', ()=>{
    attachEvents();
    interceptDesktopClicks();
  });
})();
